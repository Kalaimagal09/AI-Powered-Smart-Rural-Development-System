CREATE VIEW Executive_Infrastructure_Summary AS
SELECT 
    State,
    COUNT(Village_ID) AS Total_Villages_Tracked,
    ROUND(AVG(Electrification_Pct), 1) AS Avg_Electrification_Pct,
    ROUND(AVG(Internet_Penetration_Pct), 1) AS Avg_Internet_Pct,
    SUM(Primary_Health_Centres) AS Total_Health_Centres
FROM rural_data
GROUP BY State;


SELECT * FROM Executive_Infrastructure_Summary;
